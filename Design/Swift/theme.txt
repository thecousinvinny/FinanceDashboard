// MARK: - View modifiers

struct GlassCard: ViewModifier {
    var padding: CGFloat = 20
    var cornerRadius: CGFloat = 24
    func body(content: Content) -> some View {
        content
            .padding(padding)
            .background(
                RoundedRectangle(cornerRadius: cornerRadius, style: .continuous)
                    .fill(LumenTheme.surface)
            )
            .overlay(
                RoundedRectangle(cornerRadius: cornerRadius, style: .continuous)
                    .strokeBorder(LumenTheme.hairline, lineWidth: 1)
            )
    }
}

extension View {
    func glassCard(padding: CGFloat = 20, cornerRadius: CGFloat = 24) -> some View {
        modifier(GlassCard(padding: padding, cornerRadius: cornerRadius))
    }
}

// MARK: - Display number formatting

enum Money {
    static let formatter: NumberFormatter = {
        let f = NumberFormatter()
        f.numberStyle = .currency
        f.currencyCode = "USD"
        f.maximumFractionDigits = 2
        return f
    }()

    static func format(_ value: Double, sign: Bool = false) -> String {
        let s = formatter.string(from: NSNumber(value: abs(value))) ?? "$0"
        if sign {
            return (value < 0 ? "-" : "+") + s
        }
        return s
    }

    static func compact(_ value: Double) -> String {
        let abs = Swift.abs(value)
        let sign = value < 0 ? "-" : ""
        if abs >= 1_000_000 { return "\(sign)$\(String(format: "%.1f", abs/1_000_000))M" }
        if abs >= 10_000 { return "\(sign)$\(String(format: "%.1f", abs/1_000))K" }
        if abs >= 1_000 { return "\(sign)$\(String(format: "%.2f", abs/1_000))K" }
        return format(value)
    }
}
