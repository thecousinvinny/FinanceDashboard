import Foundation
import SwiftData

enum AccountType: String, Codable, CaseIterable, Identifiable {
    case checking, savings, investment
    var id: String { rawValue }
    var label: String { rawValue.capitalized }
}

@Model
final class BankAccount {
    var name: String
    var balance: Double
    var typeRaw: String
    var accountMask: String     // last 4 digits
    var colorHex: String
    var createdAt: Date

    init(name: String, balance: Double, type: AccountType, accountMask: String = "0000", colorHex: String = "E8C46B") {
        self.name = name
        self.balance = balance
        self.typeRaw = type.rawValue
        self.accountMask = accountMask
        self.colorHex = colorHex
        self.createdAt = Date()
    }

    var type: AccountType { AccountType(rawValue: typeRaw) ?? .checking }
}
